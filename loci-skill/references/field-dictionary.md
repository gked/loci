# Loci Field Dictionary
# Defines all frontmatter fields per type — consulted by the save skill at ingestion.
# Edit this file to evolve the schema without touching the skill.
# schema_version: 1
#
# Format:
#   field: <name>
#   type: string | array | enum | boolean | datetime | object
#   values: (for enum) pipe-separated valid values
#   required: yes | no
#   description: brief explanation

---

## Base Schema (all types)

field: uid
type: string
required: yes
description: 8-character hex identifier, generated by loci__generate_uid

field: type
type: enum
values: document | article | bookmark | conversation | project | task | event | entity | procedure | topic
required: yes
description: canonical type; determines folder and extra fields

field: title
type: string
required: yes
description: concise, descriptive title

field: author
type: string
required: yes
description: "self" for user content; entity name or UID for external content

field: created_at
type: datetime
required: yes
description: ISO 8601 timestamp of creation

field: updated_at
type: datetime
required: yes
description: ISO 8601 timestamp; bumped on every patch

field: archived_at
type: datetime
required: no
description: ISO 8601 timestamp if archived; null otherwise

field: tags
type: array
required: yes
description: 3–6 freeform lowercase hyphenated tags; consult tag-dictionary before adding new ones

field: domains
type: array
values: technology-engineering | science-research | business-strategy | finance-economics | health-biology | philosophy-ethics | creativity-design | personal-development | current-events-society
required: yes
description: one or more domain categories from fixed taxonomy

field: summary
type: string
required: yes
description: one paragraph — what this is, why it matters, key insight

field: relations
type: array
required: yes
description: links to other vault items; each entry is {uid, type} where type is a relation verb (e.g. influences, belongs-to, authored-by)

field: references
type: array
required: yes
description: external sources; each entry is {url, title, source_type}

### entity
field: subtype
type: enum
values: person | organization | place
required: yes
description: what kind of real-world actor this is; organization covers companies, NGOs, institutions

field: roles
type: array
required: no
description: one or more current roles this entity plays (e.g. [Researcher, Author, Speaker]); freeform, capitalize each role

field: organization
type: string
required: no
description: current primary affiliated organization name or UID; when updated, archive the old value to affiliations

field: location
type: string
required: no
description: current location (city, country, or freeform); when updated, archive the old value to history

field: website
type: string
required: no
description: current primary URL; when updated, archive the old value to history

field: industry
type: string
required: no
description: industry or field (for companies/organizations)

field: founded_at
type: datetime
required: no
description: founding date (for companies/organizations)

field: affiliations
type: array
required: no
description: >
  structured history of organizational relationships; each entry is:
    {organization, roles, from, to}
  where organization is a name or UID, roles is an array, from/to are YYYY or YYYY-MM-DD, and to is null if current.
  Owns all org+role history — do NOT duplicate into history[].
  Example:
    - organization: University of Sussex
      roles: [Researcher]
      from: 1995
      to: 2001
    - organization: University of Manchester
      roles: [Professor]
      from: 2001
      to: null

field: history
type: array
required: no
description: >
  general audit log for any field that changes over time, except organization (owned by affiliations).
  Each entry is: {field, value, from, to}
  where value is the old value, from/to are YYYY or YYYY-MM-DD, and to is null if still current.
  Example:
    - field: location
      value: Brighton, UK
      from: 1995
      to: 2001
    - field: website
      value: https://old-site.example.com
      from: 2000
      to: 2010
